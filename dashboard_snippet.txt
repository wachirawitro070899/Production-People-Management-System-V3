function liveAttendanceState(emp){
 // Organization Chart แสดงกะของ "วันนี้" เป็นหลัก เพื่อไม่ตีพนักงานกะดึกเป็นขาดงานตั้งแต่ช่วงเช้า
 // กรณีหลังเที่ยงคืน ถ้ามีพนักงานเช็คอินกะดึกของวันก่อนจริง จึงแสดงสถานะของ record นั้นต่อเนื่อง
 const today=thaiDateKey(),prev=dateOffsetKey(-1),clock=currentThaiMinutes();
 const prevRec=clock<720&&employeeShiftKey(emp,prev)==='night'?attendanceFor(emp.id,prev):null;
 if(prevRec?.checkIn&&!prevRec?.checkOut)return {key:'working',label:'ทำงาน • กะดึก',detail:thaiTime(new Date(prevRec.checkIn))};
 const date=today,rec=attendanceFor(emp.id,date),sh=shiftConfig(emp,date),open=timeMinutes(sh.checkInOpen),late=timeMinutes(sh.lateReasonAfter),now=clock;
 if(rec?.checkIn)return {key:'working',label:sh.key==='night'?'ทำงาน • กะดึก':'ทำงาน',detail:thaiTime(new Date(rec.checkIn))};
 if(rec?.exception?.type==='leave')return {key:'leave',label:rec.exception.leaveType==='sick'?'ลาป่วย':'ลากิจ',detail:sh.key==='night'?'กะดึก':'กะเช้า'};
 // กะดึกยังไม่ถึงเวลาเปิดเช็คชื่อ: ห้ามขึ้นขาดงาน แม้มี record เก่าที่เคยถูกสร้างผิด
 if(sh.key==='night'&&now<open)return {key:'offshift',label:'กะดึก • รอเข้ากะ',detail:`เช็คชื่อ ${sh.checkInOpen}`};
 if(rec?.exception?.type==='absent')return {key:'absent',label:'ขาดงาน',detail:sh.key==='night'?'กะดึก • ไม่มีเช็คชื่อ':'ไม่มีเช็คชื่อ'};
 if(now>late)return {key:'absent',label:'ขาดงาน',detail:sh.key==='night'?'กะดึก • ยังไม่เช็คชื่อ':'ยังไม่เช็คชื่อ'};
 if(now>=open)return {key:'waiting',label:'รอเช็คชื่อ',detail:`${sh.key==='night'?'กะดึก • ':''}ก่อน ${sh.lateReasonAfter}`};
 return {key:'offshift',label:sh.key==='night'?'กะดึก • รอเข้ากะ':'รอเข้ากะ',detail:sh.checkInOpen};
}
